// 3. MongoDB Queries

    //	1. Find all the topics and tasks which are taught in the month of October
    //	We need to query two collections separately.
        
    //	Query for Topics:
        
        db.topics.find({
            date_taught: {
                $gte: ISODate("2020-10-01T00:00:00Z"),
                $lte: ISODate("2020-10-31T23:59:59Z")
            }
        })
        
    //	Query for Tasks:
        
        db.tasks.find({
            date_assigned: {
                $gte: ISODate("2020-10-01T00:00:00Z"),
                $lte: ISODate("2020-10-31T23:59:59Z")
            }
        })

    //--- Expected Result: ---

            // Topics in Oct: HTML Basics, CSS Flexbox, Bootstrap, JS Intro, React Hooks.
            // Tasks in Oct: JS Variables(Arun), JS Variables(Meena), React Props(John), React States(Sarah).

    //***********************************************************************************************************************************//
        
    //	2. Find all the company drives which appeared between 15-Oct-2020 and 31-Oct-2020
        
        db.company_drives.find({
            drive_date: {
                $gte: ISODate("2020-10-15T00:00:00Z"),
                $lte: ISODate("2020-10-31T23:59:59Z")
            }
        })

    //--- Expected Result: ---

            // Drives (15-31 Oct): Google, Amazon. (Microsoft is excluded).

    //***********************************************************************************************************************************//
        
    //	3. Find all the company drives and students who appeared for the placement
    //	This query retrieves the company name and the list of student IDs who appeared.
        
        db.company_drives.aggregate([
            {
                $lookup: {
                    from: "users",
                    localField: "students_appeared",
                    foreignField: "_id",
                    as: "student_details"
                }
            },
            {
                $project: {
                    _id: 0,
                    company_name: 1,
                    "student_details.name": 1,
                    "student_details.email": 1
                }
            }
        ])

    //--- Expected Result: ---

    //      Drives & Students:
    //          Google: Arun and John.
    //          Amazon: Sarah and Deepa.
    //          Microsoft: Meena. 
    
    //***********************************************************************************************************************************//
        
    //	4. Find the number of problems solved by the user in codekata
    //	Assuming each document in codekata represents a user's total progress.
        
        db.codekata.aggregate([
            {
                $lookup: {
                    from: "users",
                    localField: "user_id",
                    foreignField: "_id",
                    as: "user_info"
                }
            },
            {
                $project: {
                    _id: 0,
                    "user_info.name": 1,
                    problems_solved: 1
                }
            }
        ])

    //--- Expected Result: ---

    //      Codekata Problems:
    //          John: 200
    //          Deepa: 150
    //          Arun: 120
    //          Meena: 85
    //          Sarah: 45

    //***********************************************************************************************************************************//
        
    //	5. Find all the mentors who have a mentee count of more than 15
        
        db.mentors.find({
            mentees_count: { $gt: 15 }
        })

    //--- Expected Result: ---

    //      Mentors > 15 Mentees: Two results: Sangeetha (25) and Vidya (18). (Raghav is excluded).

    //***********************************************************************************************************************************//
        
    //	6. Find the number of users who are absent and task is not submitted between 15-Oct-2020 and 31-Oct-2020
    //	This requires joining attendance and tasks based on the user and filtering by date.
        
        db.attendance.aggregate([
            {
                $match: {
                    date: {
                        $gte: ISODate("2020-10-15T00:00:00Z"),
                        $lte: ISODate("2020-10-31T23:59:59Z")
                    },
                    status: "Absent"
                }
            },
            {
                $lookup: {
                    from: "tasks",
                    localField: "user_id",
                    foreignField: "user_id",
                    as: "user_tasks"
                }
            },
            {
                $match: {
                    "user_tasks": {
                        $elemMatch: {
                            date_assigned: {
                                $gte: ISODate("2020-10-15T00:00:00Z"),
                                $lte: ISODate("2020-10-31T23:59:59Z")
                            },
                            submitted: false
                        }
                    }
                }
            },
            {
                $count: "absent_and_task_not_submitted_count"
            }
        ])

    //--- Expected Result: ---

    //      Absent & No Task (15-31 Oct):
    //          Result Count: 2
    //          Who matches?
    //              Arun: Absent Oct 18 + Task Oct 18 Not Submitted.
    //              Sarah: Absent Oct 25 + Task Oct 25 Not Submitted.
